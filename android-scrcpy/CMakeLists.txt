cmake_minimum_required(VERSION 3.20)
project(android_scrcpy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(android-scrcpy WIN32 src/main.cpp)
target_compile_definitions(android-scrcpy PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(android-scrcpy PRIVATE mfplat mf mfuuid mfreadwrite ole32 comctl32 gdi32 user32 shell32)
if(MSVC)
  target_compile_options(android-scrcpy PRIVATE /W4 /EHsc)
endif()
